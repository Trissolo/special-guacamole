// hit test

function enemiesAttack()
{
    logWrite(`Nemici: ${nm}${acapo}Robustezza: ${rf}${acapo}Attacco: ${at}${acapo}`, true);
    
    for (j = 1; j <= nm; j++)
    {
        
        // probability 
        const [x, xy] = detxxy(at);

        const am = Math.min(95, Math.floor((Math.round((41 - nm) * Math.random() + 21) * xy)));

        logWrite(`\nEnemy #${j}`);
        
        let pa;
        let xf = 0;
        
        // determine pa
        if (nm >= 4)
        {
            pa = Math.floor(at / 4);
        }
        else
        {
            pa = Math.floor(at / nm ) + 10;
            
            xf = 1;
        }

        // const cm = Math.floor(100 * Math.random()) + 1;
        
        const pp = Math.floor(100 * Math.random()) + 1;
        
        // cs --> determines whether the enemy hits (cs === 1) or misses (cs === 0)
        
        const cs = (pa < pp)? 1 : 0;
        // gameLog.push(`${debugInfo} cs= ${cs}`);
        // enemy misses 
        if (cs === 0)
        {
            // enemy misses (14190)
            xf = 1;

            logWrite(`💨 L'avversario #${j} sbaglia il colpo`);

            continue;
            // cc = 0 : ifnm > 1 then cc = int(at/15) :rem delay
        }
        else
        {
            const cc = Math.floor(am / 20);

            const cm = Math.floor(100 * Math.random()) + 1;

            if (cc >= cm) 
            {
                xf = Math.min(2, nm);
            }

            if (xf !== 0)
            {
                const pall = '💥'.repeat(Math.max(1, xf));

                logWrite(`${pall} You receive ${xf} wounds`);

                rf -= xf;

                continue;
            }
            else
            {
                logWrite(`🛡️ Parried enemy hit`);
            }
        }
        
    } // end loop

}

function detxxy(at = 80)
{

    const xxy = [2, 1];

    if (at >= 100 && at < 150)
    {
        xxy[0] = 1; //[2, 1.2];
        xxy[1] = 1.2;
    }

    if (at >= 150)
    {
        // xxy = [4, 2];
        xxy[0] = 4;
        xxy[1] = 2;

    }
    if (at >= 200)
    {
        // xxy = [6, 3.5];
        xxy[0] = 6;
        xxy[1] = 3.5;
    }

    // console.log("xxy", xxy);

    return xxy;
}

function logWrite(text, clearBefore = false, newLine = true)
{
    const guiPre = document.getElementById("demo");
    if (clearBefore)
    {
    	guiPre.replaceChildren();
    }
    
    if (text)
    {
        guiPre.innerHTML += `${text}`;
    }
    
    if (newLine)
    {
    	guiPre.innerHTML += `<br>`;
    }
    
}

function logDebugVar(text, elem, color = "black", newLine = false)
{

}
